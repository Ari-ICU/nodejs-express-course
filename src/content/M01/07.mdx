```javascript
// 1. Multi-Threaded Model (Traditional Blocking Server):
// Allocates 1 heavy OS thread per request; blocks waiting on I/O
function traditionalHandling() {
  console.log("Request 1 arrives -> Dedicated thread allocated (blocks on DB query)");
  // Thread sleeps/blocks here ⏳
}

// 2. Single-Threaded Event Loop (Node.js Non-blocking Model):
// 1 main thread processes requests; delegates long tasks without blocking
console.log("1. Request 1 arrives -> Main thread accepts connection");

setTimeout(() => {
  console.log("3. Database response ready -> Callback executed in Event Loop ⚡");
}, 100);

console.log("2. Request 2 arrives -> Main thread immediately accepts it!");

// Concurrency Comparison:
// Apache/PHP: 10,000 clients ≈ ~20 GB RAM (Thread overhead)
// Node.js:    10,000 clients ≈ ~35 MB RAM (Event-driven efficiency)
```
