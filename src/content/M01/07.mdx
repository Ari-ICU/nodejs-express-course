// 1. Multi-Threaded Model (Apache / Traditional Java):
// Allocates 1 heavy OS thread (~2MB RAM) per request; blocks on I/O
class TraditionalWorkerPool {
  handleRequest(req) {
    const thread = OS.spawnThread();       // ~2MB RAM stack allocated
    const result = thread.blockingQuery(); // Thread blocked waiting for disk/DB ⏳
    return req.send(result);
  }
}

// 2. Single-Threaded Event Loop (Node.js + Libuv):
// 1 main thread for execution; I/O offloaded asynchronously
const http = require('node:http');

const server = http.createServer(async (req, res) => {
  // Non-blocking I/O delegated to Libuv thread pool & OS kernel
  const result = await db.queryAsync('SELECT * FROM users'); // Main thread free ⚡
  res.writeHead(200, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ status: 'ok', result }));
});
