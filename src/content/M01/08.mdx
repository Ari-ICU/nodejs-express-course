```javascript
import { EventEmitter } from "node:events";

// Central Domain Event Bus
const orderEmitter = new EventEmitter();

// Subscriber 1: Notification Service
orderEmitter.on("order:placed", (order) => {
  console.log(`📧 [Email] Confirmation sent to customer for order #${order.id}`);
});

// Subscriber 2: Inventory & Warehouse
orderEmitter.on("order:placed", (order) => {
  console.log(`📦 [Warehouse] SKU reserved for item "${order.item}"`);
});

// Subscriber 3: Analytics & Revenue Telemetry
orderEmitter.on("order:placed", (order) => {
  console.log(`📊 [Analytics] Conversion recorded: $${order.price}`);
});

// Producer: Dispatches event without waiting or coupling to consumers
orderEmitter.emit("order:placed", { id: 9812, item: "MacBook Pro", price: 2499 });
```
