```javascript
// Event-Driven Pattern (Publish-Subscribe Concept):
// Components listen for named events and react when triggered

const eventListeners = {};

function on(event, handler) {
  if (!eventListeners[event]) eventListeners[event] = [];
  eventListeners[event].push(handler);
}

function emit(event, data) {
  const handlers = eventListeners[event] || [];
  handlers.forEach((fn) => fn(data));
}

// 1. Register subscribers (listeners)
on("order:placed", (order) => {
  console.log(`📧 [Email] Sending confirmation for order #${order.id}`);
});

on("order:placed", (order) => {
  console.log(`📦 [Warehouse] Reserving item "${order.item}"`);
});

// 2. Trigger the event (publisher)
console.log("Customer completes checkout...");
emit("order:placed", { id: 101, item: "Laptop", price: 1200 });
```
