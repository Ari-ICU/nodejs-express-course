```javascript
import fs from "node:fs/promises";

// Write a file
await fs.writeFile("app.log", "System started successfully\n", "utf-8");

// Append data
await fs.appendFile("app.log", "User logged in\n");

// Read content
const content = await fs.readFile("app.log", "utf-8");
console.log(content);
```
