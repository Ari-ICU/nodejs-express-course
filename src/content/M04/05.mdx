```javascript
async function fetchDashboardData(userId) {
  try {
    // Run both queries concurrently!
    const [user, orders] = await Promise.all([
      db.users.findById(userId),
      db.orders.find({ userId })
    ]);
    return { user, orders };
  } catch (error) {
    console.error("Dashboard query failed:", error);
    throw error;
  }
}
```
