```javascript
let body = "";

req.on("data", (chunk) => {
  body += chunk.toString();
});

req.on("end", () => {
  try {
    const parsedData = JSON.parse(body);
    console.log("Received POST JSON:", parsedData);
  } catch (err) {
    res.writeHead(400);
    res.end("Invalid JSON payload");
  }
});
```
