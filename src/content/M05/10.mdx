```javascript
import http from "node:http";

http.createServer((req, res) => {
  if (req.method === "POST") {
    let body = "";
    req.on("data", chunk => {
      body += chunk.toString(); // Collect stream chunks
    });
    req.on("end", () => {
      console.log("Complete Body received:", body);
      res.end("Body parsed successfully");
    });
  }
});
```
