```javascript
// routes/product.routes.js
import { Router } from "express";
const router = Router();

router.route("/")
  .get(getProducts)
  .post(createProduct);

router.route("/:id")
  .get(getProductById)
  .put(updateProduct)
  .delete(deleteProduct);

export default router;

// app.js
app.use("/api/v1/products", productRoutes);
```
