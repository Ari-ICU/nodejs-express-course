```javascript
// Authentication Guard Middleware
// Inspects request headers, validates credentials, and attaches user context

export const authenticate = (req, res, next) => {
  const authHeader = req.headers.authorization;
  if (!authHeader?.startsWith("Bearer ")) {
    return res.status(401).json({ success: false, error: "Authentication token required" });
  }

  const token = authHeader.split(" ")[1];

  // Verify token credentials
  if (token !== process.env.API_SECRET_KEY) {
    return res.status(401).json({ success: false, error: "Invalid or expired token" });
  }

  // Attach authenticated identity to request for downstream route handlers
  req.user = { id: "usr_101", role: "admin", name: "Sophea" };
  next();
};
```
