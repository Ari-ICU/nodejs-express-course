/**

const express = require("express");
const app = express();
app.use(express.json());

// In-memory database store
let articles = [
  { id: "1", title: "Mastering Node.js", author: "Dara", views: 120 },
];

// 1. READ: Collection (Safe & Idempotent)
app.get("/api/v1/articles", (req, res) => {
  res.json({ success: true, count: articles.length, data: articles });
});

// 2. CREATE: Item (Unsafe & Non-Idempotent)
app.post("/api/v1/articles", (req, res) => {
  const { title, author } = req.body;
  const newArticle = { id: String(Date.now()), title, author, views: 0 };
  articles.push(newArticle);
  res.status(201).json({ success: true, data: newArticle });
});

// 3. READ: Single Item (Safe & Idempotent)
app.get("/api/v1/articles/:id", (req, res) => {
  const item = articles.find((a) => a.id === req.params.id);
  if (!item) return res.status(404).json({ success: false, error: "Not Found" });
  res.json({ success: true, data: item });
});

// 4. UPDATE: Partial Update (Idempotent implementation)
app.patch("/api/v1/articles/:id", (req, res) => {
  const item = articles.find((a) => a.id === req.params.id);
  if (!item) return res.status(404).json({ success: false, error: "Not Found" });

  res.json({ success: true, data: item });

// 5. DELETE: Remove Item (Idempotent)
app.delete("/api/v1/articles/:id", (req, res) => {
  const idx = articles.findIndex((a) => a.id === req.params.id);
  if (idx !== -1) articles.splice(idx, 1);
  res.status(204).send(); // 204 No Content
});
