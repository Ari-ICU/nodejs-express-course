const express = require("express");
const crypto = require("crypto");
const app = express();

// Enable JSON body parsing middleware
app.use(express.json());

const products = [];

/**
 * POST /api/v1/products
 * Creates a new product entity
 */
app.post("/api/v1/products", (req, res) => {
  const { name, category, price, inStock } = req.body;

  // 1. Strict Payload Validation
  const errors = [];
  if (!name || typeof name !== "string" || name.trim().length === 0) {
    errors.push("Field 'name' is required and must be a non-empty string.");
  }
  if (price === undefined || typeof price !== "number" || price < 0) {
    errors.push("Field 'price' is required and must be a non-negative number.");
  }
  if (!category || typeof category !== "string") {
    errors.push("Field 'category' is required.");
  }

  // Reject with 400 Bad Request if validation fails
  if (errors.length > 0) {
    return res.status(400).json({
      success: false,
      error: {
        code: "VALIDATION_FAILED",
        message: "Invalid product payload submitted",
        details: errors,
      },
    });
  }

  // 2. Generate Unique Resource Identifier & Entity
  const newProduct = {
    id: crypto.randomUUID(),
    name: name.trim(),
    category: category.trim().toLowerCase(),
    price: Number(price.toFixed(2)),
    inStock: inStock ?? true,
    createdAt: new Date().toISOString(),
  };

  // 3. Persist Resource
  products.push(newProduct);

  // 4. Set Location Header & Return 201 Created
  const canonicalUrl = `/api/v1/products/${newProduct.id}`;
  res.setHeader("Location", canonicalUrl);

  res.status(201).json({
    success: true,
    data: newProduct,
  });
});

app.listen(3000, () => console.log("POST Lab running on port 3000"));
