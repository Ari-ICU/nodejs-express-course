* **204 No Content**៖ ជាស្តង់ដារឧស្សាហកម្មទូទៅបំផុត នៅពេលការលុបទិន្នន័យទទួលបានជោគជ័យ ហើយ Server មិនចាំបាច់បញ្ជូន Response Body ត្រឡប់មកវិញឡើយ (\`res.status(204).send()\`)។
* **200 OK**៖ ប្រើប្រាស់នៅពេល Server ចង់បញ្ជូន Message បញ្ជាក់ ឬទិន្នន័យដែលទើបលុបត្រឡប់ទៅ Client វិញ៖ \`{ success: true, message: "Resource deleted" }\`។
* **202 Accepted**៖ ប្រើប្រាស់នៅពេលប្រតិបត្តិការលុបជាប្រភេទ Asynchronous ឬត្រូវរង់ចាំដំណើរការក្នុង Queue (ដូចជាការលុប File ធំៗពី AWS S3)។
* **404 Not Found**៖ ឆ្លើយតបនៅពេលដែល ID របស់ទិន្នន័យដែលចង់លុប មិនមានវត្តមានក្នុង Database។

const express = require("express");
const app = express();

let products = [
  { id: "p1", name: "Wireless Mouse", isDeleted: false, deletedAt: null },
  { id: "p2", name: "Mechanical Keyboard", isDeleted: false, deletedAt: null },
];

/**
 * 1. Hard Delete Route (204 No Content)
 * DELETE /api/v1/products/:id/permanent
 */
app.delete("/api/v1/products/:id/permanent", (req, res) => {
  const { id } = req.params;
  const index = products.findIndex((p) => p.id === id);

  if (index === -1) {
    return res.status(404).json({
      success: false,
      error: { code: "NOT_FOUND", message: `Product '${id}' not found` },
    });
  }

  // Remove completely from collection
  products.splice(index, 1);

  // Return standard 204 No Content (No response body)
  res.status(204).send();
});

/**
 * 2. Soft Delete Route (Preserves record for audit/recovery)
 * DELETE /api/v1/products/:id
 */
app.delete("/api/v1/products/:id", (req, res) => {
  const { id } = req.params;
  const product = products.find((p) => p.id === id && !p.isDeleted);

  if (!product) {
    return res.status(404).json({
      success: false,
      error: { code: "NOT_FOUND", message: `Active product '${id}' not found` },
    });
  }

  // Soft delete state transition
  product.isDeleted = true;
  product.deletedAt = new Date().toISOString();

  res.status(200).json({
    success: true,
    message: "Product soft-deleted successfully",
    data: { id: product.id, deletedAt: product.deletedAt },
  });
});
