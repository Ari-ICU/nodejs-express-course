/**

const express = require("express");
const app = express();
app.use(express.json());

// Simulation DB
const users = [{ id: "usr_1", email: "alice@example.com", role: "member" }];

app.post("/api/v1/users", (req, res) => {
  const { email, password } = req.body;

  // 400 Bad Request: Missing required fields
  if (!email || !password) {
    return res.status(400).json({ success: false, error: "Email and password required" });
  }

  // 409 Conflict: Resource already exists
  const existing = users.find((u) => u.email === email);
  if (existing) {
    return res.status(409).json({ success: false, error: "Email is already registered" });
  }

  const newUser = { id: `usr_${users.length + 1}`, email, role: "member" };
  users.push(newUser);

  // 201 Created: New resource persisted
  res.status(201).json({ success: true, data: newUser });
});

// Guard: 401 vs 403 demonstration
app.delete("/api/v1/system/purge", (req, res) => {
  const token = req.headers.authorization;
  // 401: Unauthenticated
  if (!token) {
    return res.status(401).json({ success: false, error: "Authentication token required" });
  }

  // 403: Authenticated, but unauthorized role
  const userRole = "member";
  if (userRole !== "admin") {
    return res.status(403).json({ success: false, error: "Admin privileges required" });
  }

  // 204 No Content
  res.status(204).send();
});
