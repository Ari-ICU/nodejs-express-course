### ១. តារាង Status Codes ចាំបាច់ក្នុង RESTful API
* **200 OK**៖ ប្រតិបត្តិការជោគជ័យទូទៅសម្រាប់ \`GET\`, \`PUT\`, និង \`PATCH\`។
* **201 Created**៖ បង្កើត Resource ថ្មីជោគជ័យតាមរយៈ \`POST\`។
* **204 No Content**៖ ប្រតិបត្តិការជោគជ័យ ប៉ុន្តែគ្មាន Response Body ត្រឡប់ទៅវិញឡើយ (\`DELETE\`)។
* **304 Not Modified**៖ ប្រើប្រាស់ជាមួយ Caching Headers (\`ETag\`, \`If-None-Match\`) ដើម្បីបញ្ជាក់ថាទិន្នន័យលើ Browser នៅថ្មីដដែល។
* **400 Bad Request**៖ Syntax នៃ JSON មិនត្រឹមត្រូវ ឬខ្វះ Data តាម Schema។
* **401 Unauthorized**៖ មិនទាន់មានការផ្ទៀងផ្ទាត់អត្តសញ្ញាណ (User មិនទាន់បាន Login)។
* **403 Forbidden**៖ បាន Login រួចហើយ ប៉ុន្តែគ្មានសិទ្ធិអនុញ្ញាត (User មិនមែនជា Admin)។
* **404 Not Found**៖ រកមិនឃើញ Resource ឬ Endpoint URL។
* **409 Conflict**៖ ទំនាស់ទិន្នន័យ (ដូចជា Email មានក្នុងប្រព័ន្ធរួចហើយ)។
* **422 Unprocessable Entity**៖ ទម្រង់ JSON ត្រឹមត្រូវ ប៉ុន្តែខុសវិធានអាជីវកម្ម (Business Validation Rules)។
* **429 Too Many Requests**៖ Client ស្នើសុំច្រើនហួសកម្រិតកំណត់ (Rate Limit Exceeded)។
* **500 Internal Server Error**៖ កំហុសបច្ចេកទេស ឬ Exception ដែលមិនបានរំពឹងទុកនៅលើ Server។`,

/**

const express = require("express");
const app = express();
app.use(express.json());

// Simulation DB
const users = [{ id: "usr_1", email: "alice@example.com", role: "member" }];

app.post("/api/v1/users", (req, res) => {
  const { email, password } = req.body;

  // 400 Bad Request: Missing required fields
  if (!email || !password) {
    return res.status(400).json({ success: false, error: "Email and password required" });
  }

  // 409 Conflict: Resource already exists
  const existing = users.find((u) => u.email === email);
  if (existing) {
    return res.status(409).json({ success: false, error: "Email is already registered" });
  }

  const newUser = { id: `usr_${users.length + 1}`, email, role: "member" };
  users.push(newUser);

  // 201 Created: New resource persisted
  res.status(201).json({ success: true, data: newUser });
});

// Guard: 401 vs 403 demonstration
app.delete("/api/v1/system/purge", (req, res) => {
  const token = req.headers.authorization;
  // 401: Unauthenticated
  if (!token) {
    return res.status(401).json({ success: false, error: "Authentication token required" });
  }

  // 403: Authenticated, but unauthorized role
  const userRole = "member";
  if (userRole !== "admin") {
    return res.status(403).json({ success: false, error: "Admin privileges required" });
  }

  // 204 No Content
  res.status(204).send();
});
