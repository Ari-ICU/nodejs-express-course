        page: Number(page),
        limit: Number(limit),
        totalItems: total,
        totalPages: Math.ceil(total / limit),
        timestamp: new Date().toISOString(),
      },
    });
  }
}

// Example: Single resource response
app.get("/api/v1/profile", (req, res) => {
