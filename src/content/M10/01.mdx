Routes handle URLs; Controllers handle HTTP request/response; Services execute business logic; Models interface with database.

```javascript
// 1. Controller: ONLY handles HTTP inputs and responses
export const registerUser = async (req, res, next) => {
  try {
    const user = await authService.createUser(req.body);
    res.status(201).json({ success: true, data: user });
  } catch (err) {
    next(err);
  }
};

// 2. Service: Pure business logic, reusable in CLI, webhooks, or cron!
export const createUser = async (userData) => {
  const existing = await User.findOne({ email: userData.email });
  if (existing) throw new BadRequestError("Email already registered");
  const hashedPassword = await bcrypt.hash(userData.password, 12);
  return await User.create({ ...userData, password: hashedPassword });
};
```
