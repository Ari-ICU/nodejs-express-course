// services/product.service.js
const products = [
  { id: "1", name: "Laptop", price: 999 },
];

export const getById = async (id) => {
  return products.find((p) => p.id === id) || null;
};

export const createProduct = async (productData) => {
  // Pure business logic: compute tax, apply domain rules, persist
  const priceWithTax = Number((productData.price * 1.1).toFixed(2));
  const newProduct = {
    id: String(Date.now()),
    ...productData,
    priceWithTax,
  };
  products.push(newProduct);
  return newProduct;
};
