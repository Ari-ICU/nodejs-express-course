// In service: Attach statusCode to Error for Express error handling
export const getProductOrThrow = async (id) => {
  const product = products.find((p) => p.id === id);
  if (!product) {
    const error = new Error("Product does not exist");
    error.statusCode = 404;
    throw error;
  }
  return product;
};
