// Reusable Error Helper Utility
export const sendError = (res, statusCode, message, errorCode = "BAD_REQUEST", details = null) => {
  return res.status(statusCode).json({
    success: false,
    error: {
      code: errorCode,
      message,
      statusCode,
      details,
      timestamp: new Date().toISOString()
    }
  });
};
