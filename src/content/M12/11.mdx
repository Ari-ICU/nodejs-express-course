const userSchema = new Schema({
  phoneNumber: {
    type: String,
    validate: {
      validator: function(v) {
        return /^[0-9]{9,10}$/.test(v); // Cambodian phone number check
      },
      message: props => `${props.value} is not a valid phone number!`
    }
  }
});
