```javascript
// Pre-save hook: automatically normalize email before saving
userSchema.pre("save", function(next) {
  if (this.isModified("email")) {
    this.email = this.email.toLowerCase().trim();
  }
  next();
});

// Post-save hook: execute logic after document is persisted
userSchema.post("save", function(doc, next) {
  console.log(`Document persisted with ID: ${doc._id}`);
  next();
});
```
