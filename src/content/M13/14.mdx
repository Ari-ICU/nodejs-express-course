// Converts ?price[gte]=100 into { price: { $gte: 100 } }
export const filterQuery = (queryObj) => {
  let queryStr = JSON.stringify(queryObj);
  queryStr = queryStr.replace(/\b(gte|gt|lte|lt|in)\b/g, match => `$${match}`);
  return JSON.parse(queryStr);
};
