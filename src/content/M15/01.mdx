Eliminate repetitive try/catch blocks with an async wrapper and centralized error handling middleware.

```javascript
// utils/AppError.js
export class AppError extends Error {
  constructor(message, statusCode) {
    super(message);
    this.statusCode = statusCode;
    this.isOperational = true; // Distinguishes bugs from operational errors
    Error.captureStackTrace(this, this.constructor);
  }
}

// middleware/error.middleware.js (must have 4 arguments!)
export const errorHandler = (err, req, res, next) => {
  const statusCode = err.statusCode || 500;
  const message = err.message || "Internal Server Error";

  res.status(statusCode).json({
    success: false,
    statusCode,
    message,
    ...(process.env.NODE_ENV === "development" && { stack: err.stack })
  });
};
```
