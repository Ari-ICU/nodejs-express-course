// Concept breakdown in Express middleware flow:
// 1. Authentication (Who are you?) -> returns 401 if unauthenticated
app.use("/api/admin/*", verifyAuthToken);

// 2. Authorization (What can you do?) -> returns 403 if unauthorized
app.delete("/api/admin/users/:id", (req, res) => {
  if (req.user.role !== "superadmin") {
    return res.status(403).json({
      status: "fail",
      message: "Forbidden: Only superadmins can delete user accounts"
