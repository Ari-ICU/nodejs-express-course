```javascript
import bcrypt from "bcryptjs";

// 1. Generate salt and hash during registration
const SALT_ROUNDS = 12;
const plainPassword = "SecurePassword2026!";

const salt = await bcrypt.genSalt(SALT_ROUNDS);
const passwordHash = await bcrypt.hash(plainPassword, salt);
console.log("Hashed Password:", passwordHash);

// 2. Verify during user login
const candidatePassword = "SecurePassword2026!";
const isMatch = await bcrypt.compare(candidatePassword, passwordHash);

if (!isMatch) {
  throw new Error("Invalid email or password");
}
console.log("Authentication successful!");
```
