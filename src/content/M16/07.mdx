```javascript
// Structure of a JWT:
// header.payload.signature
//
// 1. Header (base64url):   {"alg": "HS256", "typ": "JWT"}
// 2. Payload (base64url):  {"userId": "64f1a2b3...", "role": "user", "exp": 1773729600}
// 3. Signature:            HMACSHA256(base64Url(header) + "." + base64Url(payload), secretKey)

// Example decoding in Node.js:
import jwt from "jsonwebtoken";

const token = jwt.sign({ sub: "user_123", role: "editor" }, process.env.JWT_SECRET, {
  expiresIn: "1h",
});
console.log("Encoded JWT:", token);
```
