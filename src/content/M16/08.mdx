```javascript
import jwt from "jsonwebtoken";

export const generateAccessToken = (userId, role) => {
  return jwt.sign(
    { 
      sub: userId, 
      role: role 
    },
    process.env.JWT_ACCESS_SECRET,
    { 
      expiresIn: "15m", // Short lifespan for security
      issuer: "api.myenterprise.com",
      audience: "myenterprise-client"
    }
  );
};
```
