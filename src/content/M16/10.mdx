```javascript
export const sendRefreshTokenCookie = (res, token) => {
  const isProduction = process.env.NODE_ENV === "production";

  res.cookie("refreshToken", token, {
    httpOnly: true, // Inaccessible to JavaScript document.cookie (Defeats XSS)
    secure: isProduction, // Transmitted only over HTTPS
    sameSite: isProduction ? "strict" : "lax", // Prevents CSRF cross-origin leakage
    maxAge: 7 * 24 * 60 * 60 * 1000, // 7 days in milliseconds
    path: "/api/v1/auth", // Restricted only to auth refresh endpoints
  });
};
```
