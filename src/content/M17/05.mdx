// controllers/user.controller.js
export const getMyProfile = async (req, res, next) => {
  // Always query by req.user._id, never trust client-supplied ID for "me" routes
  res.json({
    status: "success",
    data: { user: req.user }
  });
};

export const updateMyPassword = async (req, res, next) => {
  const { currentPassword, newPassword } = req.body;
  const user = await User.findById(req.user._id).select("+password");

  if (!(await bcrypt.compare(currentPassword, user.password))) {
    throw new AppError("Current password incorrect", 400);
  }

  user.password = newPassword;
  await user.save(); // triggers pre-save bcrypt hook

  res.json({ status: "success", message: "Password updated successfully" });
};
