```javascript
import express from "express";
import helmet from "helmet";

const app = express();

// Enable all standard 15 security headers with sensible defaults
app.use(helmet());

// Or customize individual header policies:
app.use(
  helmet({
    contentSecurityPolicy: {
      directives: {
        defaultSrc: ["'self'"],
        scriptSrc: ["'self'", "https://cdn.trusted.com"],
      },
    },
    crossOriginEmbedderPolicy: false,
    hsts: {
      maxAge: 31536000, // 1 year
      includeSubDomains: true,
      preload: true,
    },
  })
);
```
