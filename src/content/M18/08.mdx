```javascript
// 1. Generate strong secret (run in terminal):
// node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"

// 2. Explicitly specify allowed algorithms during verification:
import jwt from "jsonwebtoken";

export const verifyTokenSecurely = (token) => {
  return jwt.verify(token, process.env.JWT_ACCESS_SECRET, {
    algorithms: ["HS256"], // Strictly enforce algorithm (defeats 'none' and RS/HS confusion)
    issuer: "api.myenterprise.com",
    audience: "myenterprise-client",
  });
};
```
