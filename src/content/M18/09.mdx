```javascript
// Express Cookie Security Best Practice:
res.cookie("token", value, {
  httpOnly: true,                         // Defeats XSS (No JS access)
  secure: process.env.NODE_ENV === "production", // Defeats Sniffing (HTTPS only)
  sameSite: "strict",                    // Defeats CSRF (Never sent cross-site)
  domain: ".myenterprise.com",           // Restrict to trusted subdomains
  path: "/",
  maxAge: 3600000,                       // 1 hour
});
```
