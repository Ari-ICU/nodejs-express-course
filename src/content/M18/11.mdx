```javascript
// 1. In Mongoose Schema:
const userSchema = new mongoose.Schema({
  email: { type: String, required: true },
  password: { type: String, required: true, select: false }, // Never selected by default
  passwordResetToken: { type: String, select: false },
});

// 2. In Model JSON transformation:
userSchema.set("toJSON", {
  transform: (doc, ret) => {
    delete ret.password;
    delete ret.__v;
    return ret;
  }
});
```
