```javascript
import mongoSanitize from "express-mongo-sanitize";

// Attack scenario if unsanitized:
// Attacker sends POST body: { "email": "admin@app.com", "password": { "$ne": null } }
// MongoDB interprets: Find admin where password is not null -> LOGIN BYPASSED!

// Defense: Mount express-mongo-sanitize
// Replaces prohibited characters ($, .) with underscores or removes them entirely
app.use(mongoSanitize());

// Also enforce strict string schemas with Zod:
const loginSchema = z.object({
  email: z.string().email(),
  password: z.string().min(1), // Rejects objects like {"$ne": null} with 400 Bad Request!
});
```
