```javascript
// app.js - Enterprise Security Middleware Pipeline:
import express from "express";
import helmet from "helmet";
import cors from "cors";
import mongoSanitize from "express-mongo-sanitize";
import { globalLimiter } from "./middleware/rateLimit.js";
import { errorHandler } from "./middleware/errorHandler.js";

const app = express();

// 1. HTTP Security Headers
app.use(helmet());

// 2. CORS Policy
app.use(cors({ origin: process.env.ALLOWED_ORIGINS.split(","), credentials: true }));

// 3. Rate Limiting
app.use("/api", globalLimiter);

// 4. Body Parser with Size Limits (Prevents payload DoS)
app.use(express.json({ limit: "10kb" }));

// 5. Data Sanitization against NoSQL Injection
app.use(mongoSanitize());

// 6. Central Error Handling (Never leaks stack traces in production)
app.use(errorHandler);

export default app;
```
