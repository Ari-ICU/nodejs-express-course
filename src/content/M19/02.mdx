```javascript
// Frontend form data preparation:
const formData = new FormData();
formData.append("title", "My Summer Vacation"); // Regular text field
formData.append("coverImage", fileInput.files[0]); // Binary file field

// Using fetch: DO NOT explicitly set Content-Type! Browser appends boundary automatically!
const res = await fetch("/api/v1/posts", {
  method: "POST",
  body: formData, // Browser sets: 'multipart/form-data; boundary=----WebKitFormBoundary...'
});
```
