// 1. Mongoose Model stores only URL string
const userSchema = new mongoose.Schema({
  name: String,
  avatarUrl: {
    type: String,
    default: "/images/default-avatar.png",
  },
});

// 2. Controller uploads file and persists URL:
export const updateAvatar = async (req, res, next) => {
  try {
    if (!req.file) throw new AppError("Please provide an image", 400);

    // Upload to Cloudflare R2
    const publicUrl = await uploadToR2(req.file.buffer, req.file.originalname, req.file.mimetype);

    // Persist URL in MongoDB
    const user = await User.findByIdAndUpdate(
      req.user._id,
      { avatarUrl: publicUrl },
      { new: true }
    );

    res.json({ status: "success", data: { avatarUrl: user.avatarUrl } });
  } catch (err) {
    next(err);
  }
};
