export const generateEmailTemplate = ({ title, bodyHtml, buttonText, buttonUrl }) => {
  return `
  <!DOCTYPE html>
  <html>
  <head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
  </head>
  <body style="font-family: Arial, sans-serif; background-color: #f4f6f9; margin: 0; padding: 20px;">
    <table width="100%" border="0" cellspacing="0" cellpadding="0">
      <tr>
        <td align="center">
          <table width="600" style="background-color: #ffffff; border-radius: 8px; padding: 40px; box-shadow: 0 4px 6px rgba(0,0,0,0.05);">
            <tr>
              <td>
                <h1 style="color: #1e293b; margin-top: 0;">${title}</h1>
                <div style="color: #475569; font-size: 16px; line-height: 1.6;">
                  ${bodyHtml}
                </div>
                ${buttonUrl ? `
                <div style="margin: 30px 0; text-align: center;">
                  <a href="${buttonUrl}" style="background-color: #3b82f6; color: #ffffff; padding: 12px 24px; text-decoration: none; border-radius: 6px; font-weight: bold; display: inline-block;">${buttonText}</a>
                </div>` : ""}
                <hr style="border: none; border-top: 1px solid #e2e8f0; margin: 30px 0;">
                <p style="color: #94a3b8; font-size: 12px; margin: 0;">If you did not request this email, please ignore it.</p>
              </td>
            </tr>
          </table>
        </td>
      </tr>
    </table>
  </body>
  </html>`;
};
