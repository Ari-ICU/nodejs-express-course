```javascript
// services/mail.service.js
export const sendWelcomeEmail = async (user) => {
  const html = generateEmailTemplate({
    title: `Welcome to ${process.env.APP_NAME}, ${user.name}!`,
    bodyHtml: "<p>We're thrilled to have you on board. Explore your new dashboard and start building your projects today.</p>",
    buttonText: "Go to Dashboard",
    buttonUrl: `${process.env.CLIENT_URL}/dashboard`,
  });

  await sendEmail({
    to: user.email,
    subject: `Welcome to ${process.env.APP_NAME}!`,
    text: `Hi ${user.name}, Welcome to our platform! Log in at: ${process.env.CLIENT_URL}/login`,
    html,
  });
};
```
