```javascript
// services/emailDispatcher.js
import { sendEmail } from "./email.service.js";

// Decoupled asynchronous dispatcher: runs without blocking HTTP responses
export const dispatchEmailAsync = (options) => {
  setImmediate(async () => {
    try {
      await sendEmail(options);
      console.log(`✉️ Email sent to ${options.to}`);
    } catch (err) {
      console.error(`❌ Background email failed: ${err.message}`);
    }
  });
};

// Inside Express Controller:
export const requestPasswordReset = async (req, res) => {
  const { email } = req.body;

  // Dispatch email in background without awaiting SMTP handshake
  dispatchEmailAsync({
    to: email,
    subject: "Password Reset",
    html: "<p>Click here to reset your password.</p>",
  });

  // Immediate response to user (<50ms!)
  res.json({ success: true, message: "Reset link sent to your email!" });
};
```
