```javascript
import { Queue } from "bullmq";

// 1. Initialize BullMQ Queue
export const emailQueue = new Queue("email-notifications", {
  connection: { host: process.env.REDIS_HOST, port: Number(process.env.REDIS_PORT) },
});

// 2. Dispatch email job inside Express Controller
export const triggerPasswordReset = async (req, res, next) => {
  // ... generate resetToken ...

  // Offload to BullMQ with 3 retries and exponential backoff
  await emailQueue.add("send-reset-email", {
    email: user.email,
    token: resetToken,
  }, {
    attempts: 3,
    backoff: { type: "exponential", delay: 3000 },
  });

  res.json({ status: "success", message: "Password reset link sent to your email!" });
};
```
