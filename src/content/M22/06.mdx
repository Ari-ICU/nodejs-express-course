import { describe, it, expect, vi } from "vitest";

  it("should trigger callback when user registers", () => {
    const mockNotifier = vi.fn(); // Mock function

    mockNotifier("user@example.com");

    expect(mockNotifier).toHaveBeenCalledTimes(1);
    expect(mockNotifier).toHaveBeenCalledWith("user@example.com");
