```javascript
describe("Products CRUD API", () => {
  let createdProductId = "";

  it("POST /api/products -> 201 Created", async () => {
    const res = await request(app)
      .post("/api/products")
      .send({ title: "Gaming Mouse", price: 49.99 })
      .expect(201);

    createdProductId = res.body.data.product._id;
    expect(createdProductId).toBeDefined();
  });

  it("GET /api/products/:id -> 200 OK", async () => {
    const res = await request(app)
      .get(`/api/products/${createdProductId}`)
      .expect(200);

    expect(res.body.data.product.title).toBe("Gaming Mouse");
  });

  it("DELETE /api/products/:id -> 200 OK", async () => {
    await request(app)
      .delete(`/api/products/${createdProductId}`)
      .expect(200);
  });
});
```
