```javascript
import { generateAccessToken } from "../utils/jwt.js";

describe("Protected Routes Auth Testing", () => {
  it("should reject requests without a Bearer token with 401", async () => {
    const res = await request(app)
      .get("/api/v1/users/me")
      .expect(401);

    expect(res.body.message).toMatch(/not logged in/i);
  });

  it("should permit access when a valid JWT token is provided", async () => {
    const testToken = generateAccessToken("test_user_id_123", "user");

    const res = await request(app)
      .get("/api/v1/users/me")
      .set("Authorization", `Bearer ${testToken}`)
      .expect(200);

    expect(res.body.data.user).toBeDefined();
  });
});
```
