```javascript
import { describe, it, expect, vi } from "vitest";
import * as mailService from "../services/mail.service.js";

describe("Registration with Email Mock", () => {
  it("should register user and dispatch welcome email", async () => {
    // Spy on mailService.sendEmail and mock its implementation
    const emailSpy = vi.spyOn(mailService, "sendEmail").mockResolvedValue({ success: true });

    const res = await request(app)
      .post("/api/v1/auth/register")
      .send({ name: "Alice", email: "alice@test.com", password: "Password123!" })
      .expect(201);

    // Verify email was sent with right params without actually contacting SMTP server
    expect(emailSpy).toHaveBeenCalledTimes(1);
    expect(emailSpy).toHaveBeenCalledWith(expect.objectContaining({ to: "alice@test.com" }));

    emailSpy.mockRestore(); // Restore original function
  });
});
```
