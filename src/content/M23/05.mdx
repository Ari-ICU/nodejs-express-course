```javascript
// Postman Tests Script on POST /login:
// Save JWT token automatically to environment variables!
const response = pm.response.json();

if (response.token) {
  pm.environment.set("auth_token", response.token);
  console.log("Updated auth_token variable successfully!");
}

// In Collection Root -> Authorization Tab:
// Type: Bearer Token
// Token: {{auth_token}}
```
