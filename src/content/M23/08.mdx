```javascript
// Postman Test Script using Chai BDD syntax:
pm.test("Response time is under 200ms", () => {
  pm.expect(pm.response.responseTime).to.be.below(200);
});

pm.test("Response has required schema properties", () => {
  const json = pm.response.json();
  pm.expect(json).to.have.property("status", "success");
  pm.expect(json.data).to.be.an("object");
  pm.expect(json.data.user.email).to.match(/@/);
});
```
