Attach Socket.IO to an HTTP server and broadcast real-time events to specific chat rooms or user IDs.

```javascript
import { createServer } from "node:http";
import { Server } from "socket.io";
import app from "./app.js";

const httpServer = createServer(app);
const io = new Server(httpServer, { cors: { origin: "*" } });

io.on("connection", (socket) => {
  console.log("User connected:", socket.id);

  socket.on("joinRoom", (roomId) => {
    socket.join(roomId);
  });

  socket.on("sendMessage", ({ roomId, message, sender }) => {
    // Broadcast ONLY to clients in this specific room!
    io.to(roomId).emit("newMessage", { message, sender, timestamp: new Date() });
  });
});
```
