```javascript
io.on("connection", (socket) => {
  // Join a specific chat channel or project room
  socket.on("join_room", (roomId) => {
    socket.join(roomId);
    console.log(`Socket ${socket.id} joined room: ${roomId}`);

    // Notify others in that room only
    socket.to(roomId).emit("user_joined", { userId: socket.id });
  });

  // Send message strictly to members of this room
  socket.on("chat_in_room", ({ roomId, message }) => {
    io.to(roomId).emit("receive_room_message", {
      sender: socket.id,
      message,
      sentAt: new Date(),
    });
  });
});
```
