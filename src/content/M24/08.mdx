```javascript
// Authenticate socket and join personal notification channel:
io.use((socket, next) => {
  const token = socket.handshake.auth.token;
  try {
    const user = jwt.verify(token, process.env.JWT_SECRET);
    socket.user = user;
    next();
  } catch (err) {
    next(new Error("Authentication failed"));
  }
});

io.on("connection", (socket) => {
  // Join room named after user's database ID
  socket.join(`user:${socket.user.id}`);
});

// Anywhere in your backend (e.g. Order Service):
export const notifyUserOrderShipped = (io, userId, order) => {
  io.to(`user:${userId}`).emit("notification", {
    type: "ORDER_SHIPPED",
    title: "Order on the way!",
    message: `Order #${order.id} has been dispatched.`,
  });
};
```
