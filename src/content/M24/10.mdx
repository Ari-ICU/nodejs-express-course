```javascript
// Complete Real-Time Chat Handler:
socket.on("send_chat_message", async ({ conversationId, text }, ack) => {
  try {
    // 1. Persist message in MongoDB
    const message = await Message.create({
      conversation: conversationId,
      sender: socket.user.id,
      text,
      createdAt: new Date(),
    });

    // 2. Broadcast to all members in this conversation room
    io.to(`conversation:${conversationId}`).emit("new_chat_message", message);

    // 3. Acknowledge back to sender
    if (ack) ack({ success: true, messageId: message._id });
  } catch (err) {
    if (ack) ack({ success: false, error: err.message });
  }
});
```
