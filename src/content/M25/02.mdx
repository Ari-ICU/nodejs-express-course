```javascript
// DANGEROUS: Blocks the entire server for all concurrent users!
app.get("/bad", (req, res) => {
  const data = fs.readFileSync("/huge-file.csv"); // Event loop BLOCKED!
  res.send(data);
});

// PROPER: Non-blocking asynchronous I/O allows thousands of concurrent requests
app.get("/good", async (req, res) => {
  const data = await fs.promises.readFile("/huge-file.csv"); // Event loop stays FREE!
  res.send(data);
});
```
