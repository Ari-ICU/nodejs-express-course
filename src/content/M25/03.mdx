```javascript
// SLOW: Waterfall execution (Total: ~200ms)
const user = await User.findById(userId);       // 100ms
const orders = await Order.find({ user: userId }); // 100ms

// FAST: Parallel concurrent execution (Total: ~100ms!)
const [user, orders] = await Promise.all([
  User.findById(userId),
  Order.find({ user: userId }),
]);
```
