```javascript
// 1. Offset Pagination (Good for numbered page UI):
const page = Number(req.query.page) || 1;
const limit = Number(req.query.limit) || 20;
const skip = (page - 1) * limit;
const items = await Product.find().skip(skip).limit(limit).lean();

// 2. Cursor Pagination (Optimal for Infinite Scroll & Massive Datasets):
// Query: /api/products?cursor=64f1a2b3c4...&limit=20
const { cursor, limit = 20 } = req.query;
const query = cursor ? { _id: { $lt: cursor } } : {}; // uses indexed _id directly!

const products = await Product.find(query)
  .sort({ _id: -1 })
  .limit(Number(limit))
  .lean();
```
