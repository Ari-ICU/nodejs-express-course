```javascript
import { Queue } from "bullmq";
import { redisConnection } from "../config/redis.js";

// Initialize Queue instance
export const videoQueue = new Queue("video-processing", {
  connection: redisConnection,
});

// Producer function called from Express controller:
export const enqueueVideoTranscode = async (videoId, filePath, targetResolution) => {
  const job = await videoQueue.add(
    "transcode",
    { videoId, filePath, targetResolution },
    {
      attempts: 3, // Retry up to 3 times on crash
      backoff: { type: "exponential", delay: 2000 }, // 2s, 4s, 8s
      removeOnComplete: true, // Auto-cleanup succeeded jobs
    }
  );

  console.log(`Job enqueued: ${job.id}`);
  return job.id;
};
```
