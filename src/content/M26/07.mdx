```javascript
import { Worker } from "bullmq";
import { redisConnection } from "../config/redis.js";

export const videoWorker = new Worker(
  "video-processing",
  async (job) => {
    console.log(`[Worker] Processing job ${job.id}: transcode ${job.data.videoId}`);

    // Simulate heavy CPU transcoding
    await transcodeVideoFile(job.data.filePath, job.data.targetResolution);

    console.log(`[Worker] Job ${job.id} completed successfully!`);
    return { status: "processed", path: `/videos/${job.data.videoId}.mp4` };
  },
  {
    connection: redisConnection,
    concurrency: 4, // Process 4 video jobs concurrently
  }
);
```
