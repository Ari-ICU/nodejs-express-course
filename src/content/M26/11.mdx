```javascript
// Schedule a repeatable job to run every day at midnight (UTC)
await reportQueue.add(
  "daily-sales-summary",
  { department: "ecommerce" },
  {
    repeat: {
      pattern: "0 0 * * *", // Standard Cron: minute hour day month weekday
    },
  }
);
```
