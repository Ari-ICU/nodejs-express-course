```javascript
// controllers/course.controller.js
import * as courseService from "../services/course.service.js";

export const createCourse = async (req, res, next) => {
  try {
    const course = await courseService.createCourse(req.body, req.user._id);

    res.status(201).json({
      status: "success",
      data: { course },
    });
  } catch (err) {
    next(err);
  }
};
```
