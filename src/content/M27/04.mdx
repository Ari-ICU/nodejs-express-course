// services/product.service.js
import { Product } from "../models/product.model.js";
import { AppError } from "../utils/AppError.js";

export const createProduct = async (productData, authorId) => {
  // Business rule: Check duplicate title
  const existing = await Product.findOne({ title: productData.title });
  if (existing) {
    throw new AppError("A product with this title already exists", 409);
  }

  // Calculate pricing margin or business logic
  const profitMargin = productData.price - (productData.costPrice || 0);

  const product = await Product.create({
    ...productData,
    creator: authorId,
    profitMargin,
  });

  return product;
};
