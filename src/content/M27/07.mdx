// validators/course.validator.js
import { z } from "zod";

export const createCourseSchema = z.object({
  body: z.object({
    title: z.string().min(3).max(100),
    price: z.number().nonnegative(),
    level: z.enum(["beginner", "intermediate", "advanced"]),
    category: z.string().min(1),
  }),
});
