// validators/product.validator.js
import { z } from "zod";

export const createProductSchema = z.object({
  body: z.object({
    title: z.string().min(3).max(100),
    price: z.number().positive(),
    category: z.string().min(1),
    stock: z.number().int().nonnegative().default(0),
  }),
});
