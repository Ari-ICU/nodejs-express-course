```yaml
version: "3.8"

services:
  api:
    build: .
    ports:
      - "5000:5000"
    environment:
      - NODE_ENV=production
      - MONGODB_URI=mongodb://mongo:27017/app_prod
      - REDIS_URL=redis://redis:6379
    depends_on:
      - mongo
      - redis

  mongo:
    image: mongo:7-jammy
    restart: always
    volumes:
      - mongo_data:/data/db

  redis:
    image: redis:alpine
    restart: always

volumes:
  mongo_data:
```
