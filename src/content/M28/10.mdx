```bash
# 1. Build image
docker build -t enterprise-api:1.0.0 .

# 2. Run container with restart policy and memory boundaries
docker run -d   --name enterprise-api-prod   -p 5000:5000   --memory="512m"   --restart unless-stopped   enterprise-api:1.0.0

# 3. View container health and logs
docker ps
docker logs -f enterprise-api-prod
```
