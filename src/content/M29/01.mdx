// GET /api/v1/todos?completed=true&sort=-createdAt&page=1&limit=10

export const getTodos = catchAsync(async (req, res) => {
  const { completed, sort = "-createdAt", page = 1, limit = 10 } = req.query;
  const filter = completed !== undefined ? { completed: completed === "true" } : {};

  const [todos, total] = await Promise.all([
    Todo.find(filter)
      .sort(sort)
      .skip((Number(page) - 1) * Number(limit))
      .limit(Number(limit)),
    Todo.countDocuments(filter)
  ]);

  res.json({
    success: true,
    count: todos.length,
    total,
    page: Number(page),
    pages: Math.ceil(total / Number(limit)),
    data: todos
  });
});
