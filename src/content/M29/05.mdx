// Checkout with MongoDB Transaction:
export const checkout = async (userId, items) => {
  const session = await mongoose.startSession();
  return session.withTransaction(async () => {
    // 1. Decrement stock atomically
    for (const item of items) {
      const product = await Product.findOneAndUpdate(
        { _id: item.productId, stock: { $gte: item.quantity } },
        { $inc: { stock: -item.quantity } },
        { session, new: true }
      );
      if (!product) throw new AppError(`Insufficient stock for ${item.title}`, 400);
    }
    // 2. Create order document & trigger email job
    const order = await Order.create([{ user: userId, items }], { session });
    await emailQueue.add("order-confirmation", { orderId: order[0]._id });
    return order[0];
  });
};
